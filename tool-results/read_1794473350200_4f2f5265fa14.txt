    92→export function clientSafeError(msg?: string | null): string {
    93→  if (!msg || !msg.trim()) return "تعذر إكمال المهمة — ستتم إعادة المحاولة تلقائيًا.";

    96→  }
    97→  return msg;
    98→}

   115→/**
   116→ * تجربة المرشحين بالترتيب حتى أول نجاح.
   117→ * triedKeyIds: مفاتيح جُرِّبت داخل نفس سلسلة التنفيذ فتُتخطى (التبديل للأمام فقط).
   118→ * كل محاولة تُسجَّل في attempts_json + عدّادات المفتاح.
   119→ */
   120→async function attemptCreation(
   121→  taskId: string,
   122→  model: ModelDef,
   123→  input: Record<string, unknown>,
   124→  triedKeyIds: Set<string>
   125→): Promise<CreateOutcome> {
   126→  const pool = await getKeyPool(model.providers);
   127→  // مزود بلا مفتاح (pollinations): يُضاف مصطنعًا في **آخر** الترتيب التسلسلي
   128→  // ما لم يوجد مفتاح حقيقي له في القاعدة (لو تغيّرت سياستهم مستقبلًا وتطلب توكن)
   129→  for (const p of model.providers) {
   130→    if (!KEYLESS_PROVIDERS.has(p)) continue;
   131→    if (pool.some((c) => c.provider === p)) continue;
   132→    pool.push(syntheticKeyCandidate(p, pool.length));
   133→  }
   134→  for (const cand of pool) {
   135→    if (triedKeyIds.has(cand.key.id)) continue;
   136→    const adapter = ADAPTERS[cand.provider];
   137→    if (!adapter) continue;
   138→    const providerModel = model.providerModel[cand.provider as ProviderId] ?? "";
   139→    const keyRef: ProviderKeyRef = {
   140→      id: cand.key.id,
   141→      api_key: cand.key.api_key,
   142→      label: cand.key.label || "",
   143→    };
   144→
   145→    // تسجيل المحاولة دائمًا — نجاحًا أو فشلًا
   146→    const attempt: TaskAttempt = {
   147→      provider: cand.provider,
   148→      keyId: cand.key.id,
   149→      keyLabel: cand.key.label || "",
   150→      ok: false,
   151→      at: new Date().toISOString(),
   152→    };
   153→
   154→    try {
   155→      if (model.type === "image" && adapter.createImageSync) {
   156→        // صور متزامنة (CometAPI / Pollinations): النتيجة فورية
   157→        const { urls } = await adapter.createImageSync(keyRef, providerModel, input as never);
   158→        attempt.ok = true;
   159→        await updateTask(taskId, { attempts: [attempt] });
   160→        if (!cand.synthetic) await recordKeySuccess(cand.key);
   161→        return { ok: true, provider: cand.provider as TaskProvider, keyId: cand.key.id, syncUrls: urls };
   162→      }
   163→      const { providerTaskId } = await adapter.createTask(keyRef, providerModel, input as never);
   164→      attempt.ok = true;
   165→      await updateTask(taskId, {
   166→        attempts: [attempt],
   167→        provider: cand.provider,
   168→        provider_task_id: providerTaskId,
   169→        key_id: cand.key.id,
   170→        status: "processing",
   171→        progress: 5,
   172→      });
   173→      if (!cand.synthetic) await recordKeySuccess(cand.key);
   174→      return { ok: true, provider: cand.provider as TaskProvider, keyId: cand.key.id };
   175→    } catch (e) {
   176→      const pe = e as ProviderError;
   177→      const errClass: ErrorClass =
   178→        pe && typeof pe === "object" && "class" in pe ? (pe.class as ErrorClass) : classifyProviderError(0, String(pe));
   179→      const msg = pe instanceof Error ? pe.message : String(pe);
   180→      attempt.ok = false;
   181→      attempt.errorClass = errClass;
   182→      attempt.error = msg.slice(0, 300);
   183→      await updateTask(taskId, { attempts: [attempt] });
   184→      if (!cand.synthetic) await recordKeyFailure(cand.key, errClass, msg);
   185→
   186→      if (errClass === "params") {
   187→        // خطأ معطيات — تبديل المفاتيح لن يحلّه: توقف فورًا بفشل حقيقي
   188→        await updateTask(taskId, {
   189→          status: "failed",
   190→          error_message: clientSafeError(msg),
   191→        });
   192→        return { ok: false };
   193→      }
   194→      // disabled / exhausted / provider / temporary ⇒ الانتقال للمفتاح التالي بصمت
   195→      triedKeyIds.add(cand.key.id);
   196→      continue;
   197→    }
   198→  }
   199→  return { ok: false };
   200→}

   205→
   206→async function pollOnce(
   207→  task: TaskRow,
   208→  model: ModelDef
   209→): Promise<{ done: boolean; failed: boolean; urls?: string[] }> {
   210→  if (!task.provider_task_id || !task.key_id) return { done: false, failed: false };
   211→  const keyRow = await getApiKey(task.key_id);
   212→  if (!keyRow) return { done: false, failed: false };
   213→  const adapter = ADAPTERS[task.provider ?? ""];
   214→  if (!adapter) return { done: false, failed: false };
   215→  const providerModel = model.providerModel[task.provider as ProviderId] ?? "";
   216→
   217→  try {
   218→    const r = await adapter.checkTask(
   219→      { id: keyRow.id, api_key: keyRow.api_key, label: keyRow.label || "" },
   220→      task.provider_task_id,
   221→      providerModel
   222→    );
   223→    if (r.status === "completed" && r.urls?.length) {
   224→      await recordKeySuccess(keyRow);
   225→      await updateTask(task.id, {
   226→        status: "completed",
   227→        progress: 100,
   228→        result_urls: r.urls,
   229→      });
   230→      return { done: true, failed: false, urls: r.urls };
   231→    }
   232→    await updateTask(task.id, { progress: r.progress ?? task.progress });
   233→    return { done: false, failed: false };
   234→  } catch (e) {
   235→    const pe = e as ProviderError;
   236→    const msg = pe instanceof Error ? pe.message : String(pe);
   237→    const errClass: ErrorClass =
   238→      pe && typeof pe === "object" && "class" in pe ? (pe.class as ErrorClass) : classifyProviderError(0, msg);
   239→
   240→    // فشل منتصف المهمة برسالة بنمط مفتاح/رصيد ⇒ تبديل صامت (4.3.2)
   241→    const isKeyLike = KEY_FAIL_PATTERN.test(msg) || errClass === "disabled" || errClass === "exhausted";
   242→    if (isKeyLike) {
   243→      await recordKeyFailure(keyRow, errClass === "provider" ? "temporary" : errClass, msg);
   244→      // إزالة الربط الحالي لإعادة الإنشاء على مفتاح آخر بنفس task_id
   245→      await updateTask(task.id, {
   246→        provider: null,
   247→        provider_task_id: null,
   248→        key_id: null,
   249→        status: "processing",
   250→        error_message: "بانتظار توفر مفتاح صالح…",
   251→      });
   252→      return { done: false, failed: false, urls: undefined };
   253→    }
   254→    if (errClass === "params") {
   255→      await updateTask(task.id, { status: "failed", error_message: clientSafeError(msg) });
   256→      return { done: false, failed: true };
   257→    }
   258→    if (errClass === "provider") {
   259→      // فشل حقيقي من المزود (سياسة/محتوى/موديل) وليس مشكلة مفتاح
   260→      await updateTask(task.id, { status: "failed", error_message: clientSafeError(msg) });
   261→      return { done: false, failed: true };
   262→    }
   263→    // temporary — نستمر بالاستعلام
   264→    return { done: false, failed: false };
   265→  }
   266→}

   268→/**
   269→ * التوليد: تسجيل المهمة ثم تجربة المفاتيح بالترتيب ثم نافذة انتظار ببث تقدم.
   270→ * يعيد المهمة النهائية (مكتملة) أو مهمة processing يتابعها العميل بـ get_task.
   271→ */
   272→export async function runGeneration(opts: {
   273→  model: ModelDef;
   274→  params: {
   275→    prompt: string;
   276→    aspect_ratio?: string;
   277→    resolution?: string;
   278→    background?: string;
   279→    output_format?: string;
   280→    image_url?: string;
   281→    enable_audio?: boolean;
   282→    duration?: number;
   283→  };
   284→  wait_seconds?: number;
   285→  onProgress?: ProgressCb;
   286→}): Promise<TaskRow> {
   287→  const { model, params } = opts;
   288→  const defaultWait = model.type === "image" ? 90 : 25;
   289→  const waitSeconds = clampWait(opts.wait_seconds, defaultWait);
   290→
   291→  // بناء input المعطيات الأصلية (تُخزَّن لإعادة الإنشاء عند التبديل)
   292→  const input: Record<string, unknown> = { prompt: params.prompt };
   293→  if (params.aspect_ratio) input.aspect_ratio = params.aspect_ratio;
   294→  if (params.resolution) input.resolution = params.resolution;
   295→  if (params.background) input.background = params.background;
   296→  if (params.output_format) input.output_format = params.output_format;
   297→  if (params.image_url) input.image_url = params.image_url;
   298→  if (typeof params.enable_audio === "boolean") input.enable_audio = params.enable_audio;
   299→  if (typeof params.duration === "number") input.duration = params.duration;
   300→
   301→  // تسجيل المهمة قبل أي نداء خارجي (حتى لو انقطع الاتصال تبقى قابلة للتتبع)
   302→  const taskId = randomUUID();
   303→  await insertTask({
   304→    id: taskId,
   305→    model: model.canonical,
   306→    type: model.type,
   307→    prompt: params.prompt,
   308→    params: input,
   309→    status: "pending",
   310→  });
   311→
   312→  // تجربة المفاتيح بالترتيب التسلسلي الصارم
   313→  const tried = new Set<string>();
   314→  const outcome = await attemptCreation(taskId, model, input, tried);
   315→
   316→  if (!outcome.ok) {
   317→    const still = await getTask(taskId);
   318→    if (still && still.status !== "failed") {
   319→      // بقيت processing برسالة داخلية — تُعاد محاولتها مع كل get_task
   320→      await updateTask(taskId, {
   321→        status: "processing",
   322→        error_message: "بانتظار توفر مفتاح صالح…",
   323→        params_merge: { _lastCreateAttempt: new Date().toISOString() },
   324→      });
   325→    }
   326→    return (await getTask(taskId))!;
   327→  }
   328→
   329→  // صور متزامنة اكتملت فورًا
   330→  if (outcome.syncUrls?.length) {
   331→    await updateTask(taskId, {
   332→      provider: outcome.provider,
   333→      key_id: outcome.keyId,
   334→      status: "completed",
   335→      progress: 100,
   336→      result_urls: outcome.syncUrls,
   337→    });
   338→    return (await getTask(taskId))!;
   339→  }
   340→
   341→  // نافذة الانتظار: استعلام كل 3 ثوانٍ + بث تقدم
   342→  let task = await getTask(taskId)!;
   343→  const deadline = Date.now() + waitSeconds * 1000;
   344→  while (Date.now() < deadline) {
   345→    await sleep(POLL_INTERVAL_MS);
   346→    task = (await getTask(taskId))!;
   347→    if (task.status === "completed" || task.status === "failed") return task;
   348→
   349→    // لم تُنشأ لدى المزود بعد (فشل كل المفاتيح سابقًا) — نعيد المحاولة داخل النافذة
   350→    if (!task.provider_task_id) {
   351→      const again = await attemptCreation(taskId, model, input, new Set());
   352→      if (again.syncUrls?.length) {
   353→        await updateTask(taskId, { status: "completed", progress: 100, result_urls: again.syncUrls });
   354→        return (await getTask(taskId))!;
   355→      }
   356→      continue;
   357→    }
   358→
   359→    const r = await pollOnce(task, model);
   360→    if (r.done) return (await getTask(taskId))!;
   361→    if (r.failed) return (await getTask(taskId))!;
   362→
   363→    const fresh = await getTask(taskId)!;
   364→    const pct = fresh?.progress ?? 0;
   365→    await opts.onProgress?.(
   366→      pct,
   367→      fresh && !fresh.provider_task_id
   368→        ? "بانتظار توفر مفتاح صالح… جارٍ إعادة المحاولة تلقائيًا"
   369→        : `قيد التوليد… ${pct}%`
   370→    );
   371→  }
   372→  // انتهت النافذة دون نتيجة ⇒ processing ويكملها العميل بـ get_task
   373→  return (await getTask(taskId))!;
   374→}

   380→/** إعادة إنشاء مهمة بلا ربط حالي (فشلت كل المفاتيح سابقًا) بعد تهدئة 45 ثانية */
   381→async function retryPendingCreation(task: TaskRow): Promise<TaskRow> {
   382→  let meta: Record<string, unknown> = {};
   383→  try {
   384→    meta = JSON.parse(task.params_json || "{}");
   385→  } catch {
   386→    meta = {};
   387→  }
   388→  const last = typeof meta._lastCreateAttempt === "string" ? Date.parse(meta._lastCreateAttempt) : 0;
   389→  if (Date.now() - last < COOLDOWN_MS) return task; // داخل فترة التهدئة — ننتظر
   390→
   391→  const input = meta as Record<string, unknown>;
   392→  const model = modelByCanonical(task.model);
   393→  if (!model) return task;
   394→
   395→  await updateTask(task.id, { params_merge: { _lastCreateAttempt: new Date().toISOString() } });
   396→  const outcome = await attemptCreation(task.id, model, input, new Set());
   397→  if (outcome.syncUrls?.length) {
   398→    await updateTask(task.id, { status: "completed", progress: 100, result_urls: outcome.syncUrls });
   399→  }
   400→  return (await getTask(task.id))!;
   401→}

   403→export async function refreshTask(taskId: string): Promise<TaskRow | null> {
   404→  const task = await getTask(taskId);
   405→  if (!task) return null;
   406→  if (task.status === "completed" || task.status === "failed") return task;
   407→
   408→  // دمج / تعليق مصوّر: منطق merger مستقل — نفرّق بالموديل
   409→  if (task.provider === "merger") {
   410→    return task.model === SLIDESHOW_MODEL ? refreshSlideshowTask(task) : refreshMergeTask(task);
   411→  }
   412→
   413→  // لم تُنشأ بعد لدى المزود ⇒ إعادة محاولة الإنشاء (بتهدئة)
   414→  if (!task.provider_task_id || !task.key_id) {
   415→    return retryPendingCreation(task);
   416→  }
   417→
   418→  const model = modelByCanonical(task.model);
   419→  if (!model) return task;
   420→
   421→  const r = await pollOnce(task, model);
   422→  if (r.done || r.failed) return (await getTask(taskId))!;
   423→
   424→  // فشل منتصف المهمة بنمط مفتاح/رصيد ⇒ أعاد pollOnce الربط إلى null —
   425→  // نعيد الإنشاء فورًا على المفتاح التالي بنفس task_id (نتجاوز التهدئة هذه المرة
   426→  // لأن التهدئة تخص فقط إعادة المحاولة عند «عدم توفر أي مفتاح»، لا التبديل الحي)
   427→  const fresh = await getTask(taskId)!;
   428→  if (fresh && !fresh.provider_task_id) {
   429→    await updateTask(fresh.id, {
   430→      params_merge: { _lastCreateAttempt: new Date(0).toISOString() },
   431→    });
   432→    return retryPendingCreation(fresh);
   433→  }
   434→  return fresh;
   435→}

   443→
   444→async function fireProcessOnce(taskId: string, jobId: string, kind: "merge" | "slideshow"): Promise<void> {
   445→  if (processInFlight.has(jobId)) return;
   446→  processInFlight.add(jobId);
   447→  // إطلاق خلفي بمرة واحدة فقط — النتيجة تُقرأ لاحقًا من GET الحالة
   448→  void (async () => {
   449→    try {
   450→      if (kind === "slideshow") await processMergerSlideshow(jobId);
   451→      else await processMergerJob(jobId);
   452→    } catch {
   453→      // فشل process المعلن يظهر في GET status كذلك — لا نتعامل هنا
   454→    } finally {
   455→      processInFlight.delete(jobId);
   456→    }
   457→  })();
   458→  await updateTask(taskId, { params_merge: { processStarted: true } });
   459→}

   461→async function applyMergerState(taskId: string, st: MergerJobState): Promise<"done" | "failed" | "continue"> {
   462→  if (st.status === "done") {
   463→    // رابط النتيجة: videoUrl ثم downloadUrl ثم البناء من jobId (تم في merger.ts)
   464→    const url = st.videoUrl ?? st.downloadUrl;
   465→    if (!url) {
   466→      await updateTask(taskId, { status: "failed", error_message: "اكتمل الدمج دون رابط نتيجة" });
   467→      return "failed";
   468→    }
   469→    await updateTask(taskId, {
   470→      status: "completed",
   471→      progress: 100,
   472→      result_urls: [url],
   473→    });
   474→    return "done";
   475→  }
   476→  if (st.status === "failed") {
   477→    await updateTask(taskId, { status: "failed", error_message: st.error ?? "فشل الدمج" });
   478→    return "failed";
   479→  }
   480→  await updateTask(taskId, { status: "processing", progress: st.progress });
   481→  return "continue";
   482→}

   489→/**
   490→ * دمج عدة مقاطع في فيديو واحد بالترتيب المعطى.
   491→ * التحقق الصارم: ≥2 و ≤10 وروابط http/https مباشرة عامة.
   492→ * روابط task_ids تُحل أُولًا ثم urls المباشرة بنفس ترتيبها.
   493→ */
   494→export async function runMerge(opts: {
   495→  urls?: string[];
   496→  task_ids?: string[];
   497→  wait_seconds?: number;
   498→  onProgress?: ProgressCb;
   499→}): Promise<MergeOutcome> {
   500→  const urls: string[] = [];
   501→  const failedRefs: string[] = [];
   502→
   503→  // 1) حل روابط task_ids من قاعدة البيانات بالترتيب
   504→  for (const tid of opts.task_ids ?? []) {
   505→    const t = await getTask(tid.trim());
   506→    if (!t || t.status !== "completed" || !t.result_urls_json) {
   507→      failedRefs.push(`${tid}: ${!t ? "غير موجودة" : t.status === "failed" ? "فاشلة" : "غير مكتملة بعد"}`);
   508→      continue;
   509→    }
   510→    try {
   511→      const arr = JSON.parse(t.result_urls_json) as string[];
   512→      if (arr.length) urls.push(arr[0]);
   513→    } catch {
   514→      failedRefs.push(`${tid}: روابط نتيجتها تالفة`);
   515→    }
   516→  }
   517→  // 2) ثم الروابط المباشرة بنفس ترتيبها
   518→  for (const u of opts.urls ?? []) urls.push(u.trim());
   519→
   520→  if (urls.length + failedRefs.length < 2) {
   521→    return {
   522→      error:
   523→        "❌ الدمج يحتاج مقطعَين على الأقل. مرّر urls (روابط مباشرة) أو task_ids لمهام مكتملة." +
   524→        (failedRefs.length ? `\nتفاصيل: ${failedRefs.join(" | ")}` : ""),
   525→    };
   526→  }
   527→  if (failedRefs.length) {
   528→    return { error: `❌ تعذّر حل بعض المعرفات للدمج: ${failedRefs.join(" | ")}` };
   529→  }
   530→  if (urls.length > 10) {
   531→    return { error: "❌ أقصى عدد مقاطع للدمج هو 10 (خدمة الدمج تقبل 10 مقاطع كحد أقصى)." };
   532→  }
   533→  // روابط مباشرة عامة فقط
   534→  for (const u of urls) {
   535→    if (!/^https?:\/\//i.test(u)) {
   536→      return { error: `❌ كل المقاطع يجب أن تكون روابط http/https مباشرة وعامة — رفض: ${u.slice(0, 120)}` };
   537→    }
   538→  }
   539→
   540→  // تسجيل المهمة أولًا ثم النداء على الخدمة
   541→  const taskId = randomUUID();
   542→  await insertTask({
   543→    id: taskId,
   544→    provider: "merger",
   545→    model: "video-merger",
   546→    type: "merge",
   547→    prompt: `دمج ${urls.length} مقاطع`,
   548→    params: { videos: urls },
   549→    status: "processing",
   550→  });
   551→
   552→  let jobId: string;
   553→  try {
   554→    const created = await createMergerJob(urls);
   555→    jobId = created.jobId;
   556→  } catch (e) {
   557→    const pe = e as ProviderError;
   558→    const terminal = (pe as ProviderError & { mergerTerminal?: boolean }).mergerTerminal === true;
   559→    if (terminal) {
   560→      await updateTask(taskId, { status: "failed", error_message: clientSafeError(pe.message) });
   561→      return { task: await getTask(taskId) };
   562→    }
   563→    // شبكتنا نحن — تبقى processing وتُعاد المحاولة مع get_task
   564→    await updateTask(taskId, {
   565→      status: "processing",
   566→      error_message: "تعذر إنشاء مهمة الدمج الآن — ستُعاد المحاولة تلقائيًا مع get_task",
   567→      params_merge: { _lastCreateAttempt: new Date().toISOString(), _videos: urls },
   568→    });
   569→    return { task: await getTask(taskId) };
   570→  }
   571→
   572→  await updateTask(taskId, { provider_task_id: jobId, status: "processing", progress: 5 });
   573→  await fireProcessOnce(taskId, jobId, "merge");
   574→
   575→  // نافذة الانتظار الافتراضية 60 ثانية — استعلام كل 3 ثوانٍ وكتابة الحالة فورًا
   576→  const waitSeconds = clampWait(opts.wait_seconds, 60);
   577→  const deadline = Date.now() + waitSeconds * 1000;
   578→  while (Date.now() < deadline) {
   579→    await sleep(POLL_INTERVAL_MS);
   580→    let st: MergerJobState;
   581→    try {
   582→      st = await getMergerJob(jobId);
   583→    } catch (e) {
   584→      const pe = e as ProviderError;
   585→      if ((pe as ProviderError & { mergerTerminal?: boolean }).mergerTerminal) {
   586→        // 404/فشل معلن من الخدمة = نهائي (لكن إن لم يبدأ process بعد فقد استُدعي الآن فورًا)
   587→        await updateTask(taskId, { status: "failed", error_message: clientSafeError(pe.message) });
   588→        return { task: await getTask(taskId) };
   589→      }
   590→      continue; // شبكتنا نحن — نستمر حتى نهاية النافذة
   591→    }
   592→    const r = await applyMergerState(taskId, st);
   593→    if (r === "done" || r === "failed") return { task: await getTask(taskId) };
   594→    await opts.onProgress?.(st.progress, `جارٍ الدمج… ${st.progress}%`);
   595→  }
   596→  return { task: await getTask(taskId) };
   597→}

   599→/** تتبع مهمة دمج جارية عبر get_task (بند 7.3.6) */
   600→export async function refreshMergeTask(task: TaskRow): Promise<TaskRow> {
   601→  // لم يُنشأ jobId بعد ⇒ إعادة محاولة الإنشاء بعد تهدئة
   602→  if (!task.provider_task_id) {
   603→    let meta: Record<string, unknown> = {};
   604→    try {
   605→      meta = JSON.parse(task.params_json || "{}");
   606→    } catch {
   607→      meta = {};
   608→    }
   609→    const videos = (meta._videos as string[] | undefined) ?? (meta.videos as string[] | undefined) ?? [];
   610→    const last = typeof meta._lastCreateAttempt === "string" ? Date.parse(meta._lastCreateAttempt) : 0;
   611→    if (!videos.length || Date.now() - last < COOLDOWN_MS) return task;
   612→    try {
   613→      const created = await createMergerJob(videos);
   614→      await updateTask(task.id, {
   615→        provider_task_id: created.jobId,
   616→        status: "processing",
   617→        error_message: null,
   618→      });
   619→      await fireProcessOnce(task.id, created.jobId, "merge");
   620→    } catch {
   621→      await updateTask(task.id, { params_merge: { _lastCreateAttempt: new Date().toISOString() } });
   622→    }
   623→    return (await getTask(task.id))!;
   624→  }
   625→
   626→  const jobId = task.provider_task_id;
   627→
   628→  // إن لم يُنفَّذ process بعد فنفّذه الآن (مثلًا: انتهت النافذة الأولى قبل إطلاقه)
   629→  let meta: Record<string, unknown> = {};
   630→  try {
   631→    meta = JSON.parse(task.params_json || "{}");
   632→  } catch {
   633→    meta = {};
   634→  }
   635→  if (!meta.processStarted) await fireProcessOnce(task.id, jobId, "merge");
   636→
   637→  try {
   638→    const st = await getMergerJob(jobId);
   639→    await applyMergerState(task.id, st);
   640→  } catch (e) {
   641→    const pe = e as ProviderError;
   642→    if ((pe as ProviderError & { mergerTerminal?: boolean }).mergerTerminal) {
   643→      await updateTask(task.id, { status: "failed", error_message: clientSafeError(pe.message) });
   644→    }
   645→    // شبكتنا نحن ⇒ تبقى processing حتى الاستعلام التالي
   646→  }
   647→  return (await getTask(task.id))!;
   648→}

   692→/**
   693→ * حل عناصر الصور: الروابط المباشرة تُمرَّر كما هي، والنصوص تُولَّد كصور
   694→ * بالترتيب التسلسلي (المفاتيح أولًا ثم Pollinations المجاني أخيرًا).
   695→ * حد المعدل المجاني (صورة كل ~45 ث/‏IP) يُعالج بانتظار retryAfterMs حتى سقف waitSeconds.
   696→ */
   697→async function resolveImageUrls(
   698→  items: string[],
   699→  aspect: string,
   700→  waitSeconds: number,
   701→  onProgress?: ProgressCb
   702→): Promise<string[]> {
   703→  const imageModel = modelByCanonical("gpt-image-2-6/flare-text-to-image");
   704→  if (!imageModel) throw new Error("موديل الصور غير معرّف");
   705→  const deadline = Date.now() + waitSeconds * 1000;
   706→  const prompts = items.filter((i) => !/^https?:\/\//i.test(i)).length;
   707→  const urls: string[] = [];
   708→  let generated = 0;
   709→
   710→  for (let idx = 0; idx < items.length; idx++) {
   711→    const item = items[idx].trim();
   712→    if (/^https?:\/\//i.test(item)) {
   713→      urls.push(item);
   714→      continue;
   715→    }
   716→    let done = false;
   717→    let lastErr = "";
   718→    while (!done && Date.now() < deadline) {
   719→      const pool = await getKeyPool(imageModel.providers);
   720→      for (const p of imageModel.providers) {
   721→        if (!KEYLESS_PROVIDERS.has(p)) continue;
   722→        if (pool.some((c) => c.provider === p)) continue;
   723→        pool.push(syntheticKeyCandidate(p, pool.length));
   724→      }
   725→      let retriable = false;
   726→      for (const cand of pool) {
   727→        const adapter = ADAPTERS[cand.provider];
   728→        if (!adapter?.createImageSync) continue;
   729→        const providerModel = imageModel.providerModel[cand.provider as ProviderId] ?? "";
   730→        try {
   731→          const { urls: u } = await adapter.createImageSync(
   732→            { id: cand.key.id, api_key: cand.key.api_key, label: cand.key.label || "" },
   733→            providerModel,
   734→            { prompt: item, aspect_ratio: aspect }
   735→          );
   736→          if (u.length) {
   737→            urls.push(u[0]);
   738→            done = true;
   739→          }
   740→          break;
   741→        } catch (e) {
   742→          const pe = e as ProviderError;
   743→          const cls: ErrorClass =
   744→            pe && typeof pe === "object" && "class" in pe
   745→              ? (pe.class as ErrorClass)
   746→              : classifyProviderError(0, String(pe));
   747→          lastErr = pe instanceof Error ? pe.message : String(pe);
   748→          if (cls === "temporary") {
   749→            const wait = Math.min(
   750→              (pe as ProviderError & { retryAfterMs?: number }).retryAfterMs ?? 8_000,
   751→              Math.max(1_000, deadline - Date.now())
   752→            );
   753→            await sleep(wait);
   754→            retriable = true;
   755→            break; // أعد المحاولة من أول الترتيب بعد الانتظار
   756→          }
   757→          // disabled/exhausted/provider ⇒ المزود التالي فورًا
   758→          continue;
   759→        }
   760→      }
   761→      if (!done && !retriable && Date.now() < deadline) {
   762→        // كل المزودين فشلوا بأخطاء غير مؤقتة — لا فائدة من مواصلة الانتظار
   763→        throw new Error(`تعذر توليد الصورة ${generated + 1}: ${clientSafeError(lastErr)}`);
   764→      }
   765→    }
   766→    if (!done) {
   767→      throw new Error(
   768→        `انتهى وقت تجهيز الصور قبل الاكتمال — الوصول المجاني يسمح بصورة كل ~45 ثانية (${prompts} صور نصية ≈ ${prompts * 45} ث). ` +
   769→          `الحلول: مرّر روابط صور جاهزة بدل الأوصاف، أو قلّل عدد الصور، أو زد wait_seconds حتى 240.`
   770→      );
   771→    }
   772→    generated++;
   773→    await onProgress?.(
   774→      Math.round((generated / prompts) * 100),
   775→      `تجهيز الصور ${generated}/${prompts}…`
   776→    );
   777→  }
   778→  return urls;
   779→}

   786→/** فيديو تعليق مصوّر: صور (روابط أو أوصاف) + نص راوي أو صوت جاهز */
   787→export async function runSlideshow(opts: {
   788→  images: string[];
   789→  narration_text?: string;
   790→  audio_url?: string;
   791→  voice_lang?: string;
   792→  per_image_seconds?: number;
   793→  resolution?: string;
   794→  aspect_ratio?: string;
   795→  wait_seconds?: number;
   796→  onProgress?: ProgressCb;

   798→  const images = (opts.images ?? []).map((s) => String(s || "").trim()).filter(Boolean);
   799→  if (images.length < 1) return { error: "❌ مرّر صورة واحدة على الأقل — رابط مباشر أو وصف نصي لتوليدها." };
   800→  if (images.length > 20) return { error: "❌ أقصى عدد صور هو 20." };
   801→  const narration = (opts.narration_text || "").trim();
   802→  const audioUrl = (opts.audio_url || "").trim();
   803→  if (!narration && !audioUrl) {
   804→    return { error: "❌ مرّر نص التعليق الصوتي narration_text أو رابط صوت جاهز audio_url." };
   805→  }
   806→  if (narration && audioUrl) {
   807→    return { error: "❌ مرّر narration_text أو audio_url — ليس الاثنين معًا." };
   808→  }
   809→  if (narration.length > 8000) return { error: "❌ نص التعليق طويل جدًا (الحد 8000 حرف)." };
   810→  const aspect = ["16:9", "9:16", "1:1"].includes(opts.aspect_ratio || "") ? opts.aspect_ratio! : "16:9";
   811→  const resolution = ["480p", "720p", "1080p"].includes(opts.resolution || "") ? opts.resolution! : "720p";
   812→

   814→  await insertTask({
   815→    id: taskId,
   816→    provider: "merger",
   817→    model: SLIDESHOW_MODEL,
   818→    type: "video",
   819→    prompt: narration ? `تعليق مصوّر: ${narration.slice(0, 80)}` : `تعليق مصوّر (${images.length} صور بصوت جاهز)`,
   820→    params: { images, narration_text: narration, audio_url: audioUrl, aspect_ratio: aspect, resolution },
   821→    status: "processing",
   822→  });
   823→
   824→  // 1) تجهيز روابط الصور (توليد للنصوص)
   825→  let urls: string[];
   826→  try {
   827→    const waitSeconds = clampWait(opts.wait_seconds, 120);
   828→    urls = await resolveImageUrls(images, aspect, waitSeconds, opts.onProgress);
   829→  } catch (e) {
   830→    const msg = e instanceof Error ? e.message : String(e);
   831→    await updateTask(taskId, { status: "failed", error_message: clientSafeError(msg) });
   832→    return { task: await getTask(taskId) };
   833→  }
   834→
   835→  // 2) إنشاء المهمة في خدمة الدمج
   836→  let jobId: string;
   837→  try {
   838→    const created = await createMergerSlideshow({
   839→      images: urls,
   840→      narration_text: narration || undefined,
   841→      audio_url: audioUrl || undefined,
   842→      voice_lang: opts.voice_lang || "ar",
   843→      per_image_seconds: opts.per_image_seconds,
   844→      resolution,
   845→      aspect_ratio: aspect,
   846→    });
   847→    jobId = created.jobId;
   848→  } catch (e) {
   849→    const pe = e as ProviderError;
   850→    const terminal = (pe as ProviderError & { mergerTerminal?: boolean }).mergerTerminal === true;
   851→    if (terminal) {
   852→      await updateTask(taskId, { status: "failed", error_message: clientSafeError(pe.message) });
   853→      return { task: await getTask(taskId) };
   854→    }
   855→    await updateTask(taskId, {
   856→      status: "processing",
   857→      error_message: "تعذر إنشاء مهمة التعليق الآن — ستُعاد المحاولة تلقائيًا مع get_task",
   858→      params_merge: { _lastCreateAttempt: new Date().toISOString(), _slideshow: { images: urls, narration_text: narration, audio_url: audioUrl, voice_lang: opts.voice_lang || "ar", per_image_seconds: opts.per_image_seconds, resolution, aspect_ratio: aspect } },
   859→    });
   860→    return { task: await getTask(taskId) };
   861→  }
   862→
   863→  await updateTask(taskId, { provider_task_id: jobId, status: "processing", progress: 5 });
   864→  await fireProcessOnce(taskId, jobId, "slideshow");
   865→
   866→  // 3) نافذة انتظار مع بث التقدم
   867→  const waitSeconds = clampWait(opts.wait_seconds, 120);
   868→  const deadline = Date.now() + waitSeconds * 1000;
   869→  while (Date.now() < deadline) {
   870→    await sleep(POLL_INTERVAL_MS);
   871→    let st: MergerJobState;
   872→    try {
   873→      st = await getMergerSlideshow(jobId);
   874→    } catch (e) {
   875→      const pe = e as ProviderError;
   876→      if ((pe as ProviderError & { mergerTerminal?: boolean }).mergerTerminal) {
   877→        await updateTask(taskId, { status: "failed", error_message: clientSafeError(pe.message) });
   878→        return { task: await getTask(taskId) };
   879→      }
   880→      continue;
   881→    }
   882→    const r = await applyMergerState(taskId, st);
   883→    if (r === "done" || r === "failed") return { task: await getTask(taskId) };
   884→    await opts.onProgress?.(st.progress, `بناء الفيديو… ${st.progress}%`);
   885→  }
   886→  return { task: await getTask(taskId) };
   887→}

   890→export async function refreshSlideshowTask(task: TaskRow): Promise<TaskRow> {
   891→  if (!task.provider_task_id) {
   892→    let meta: Record<string, unknown> = {};
   893→    try {
   894→      meta = JSON.parse(task.params_json || "{}");

   896→      meta = {};
   897→    }
   898→    const ss = (meta._slideshow ?? null) as
   899→      | { images: string[]; narration_text?: string; audio_url?: string; voice_lang?: string; per_image_seconds?: number; resolution?: string; aspect_ratio?: string }
   900→      | null;
   901→    const last = typeof meta._lastCreateAttempt === "string" ? Date.parse(meta._lastCreateAttempt) : 0;
   902→    if (!ss || Date.now() - last < COOLDOWN_MS) return task;
   903→    try {
   904→      const created = await createMergerSlideshow({
   905→        images: ss.images,
   906→        narration_text: ss.narration_text || undefined,

   909→        per_image_seconds: ss.per_image_seconds,
   910→        resolution: ss.resolution || "720p",
   911→        aspect_ratio: ss.aspect_ratio || "16:9",
   912→      });
   913→      await updateTask(task.id, {
   914→        provider_task_id: created.jobId,
   915→        status: "processing",
   916→        error_message: null,
   917→      });
   918→      await fireProcessOnce(task.id, created.jobId, "slideshow");
   919→    } catch {
   920→      await updateTask(task.id, { params_merge: { _lastCreateAttempt: new Date().toISOString() } });
   921→    }
   922→    return (await getTask(task.id))!;
   923→  }
   924→
   925→  const jobId = task.provider_task_id;
   926→  let meta2: Record<string, unknown> = {};

   929→  } catch {
   930→    meta2 = {};

   936→    await applyMergerState(task.id, st);

   940→      await updateTask(task.id, { status: "failed", error_message: clientSafeError(pe.message) });
   941→    }
   942→  }
   943→  return (await getTask(task.id))!;
   944→}
