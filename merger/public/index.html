    🛠️ توثيق API للمطورين (انقر للفتح)
    MethodEndpointالوظيفة
    POST/api/jobsإنشاء مهمة دمج (روابط أو مصادر ملفات)
    POST/api/upload?job=&seq=&chunk=رفع أجزاء الملفات (ثنائي ≤4MB)
    POST/api/jobs/<id>/processتنفيذ الدمج كاملًا → رابط الفيديو النهائي
    GET/api/jobs/<id>الحالة + التقدم (0-100) + الرابط عند الاكتمال
    GET/api/jobs/<id>/videoبث الناتج بدعم Range — ?download=1 للتحميل
    DELETE/api/jobs/<id>حذف فوري شامل
    GET/api/configالحدود العامة + version
    GET/api/cronحذف المهام المنتهية (يومي)

    الأخطاء دائمًا بصيغة {"error":"..."} برسائل عربية. جميع النقاط CORS مفتوح بلا تسجيل دخول.

    curl -X POST DOMAIN/api/jobs \
  -H "Content-Type: application/json" \
  -d '{"videos":["https://example.com/clip1.mp4","https://example.com/clip2.mp4"]}'
# ← 201 {"jobId":"...","status":"queued","processUrl":"...","expiresAt":"..."}

curl -X POST DOMAIN/api/jobs/JOB_ID/process
# ← 200 {"status":"done","videoUrl":"...","size":...,"duration":...,"resolution":"WxH"}
    # متابعة الحالة ثم البث/التحميل
curl DOMAIN/api/jobs/JOB_ID
curl -H "Range: bytes=0-1023" DOMAIN/api/jobs/JOB_ID/video   # ← 206
curl -o merged.mp4 "DOMAIN/api/jobs/JOB_ID/video?download=1"
